Load exactly one build per page. Structure and theme are separate files, and plugin-core.css must come before plugin-theme.css. Your own overrides go after both.

Debugging

Option precedence and the HTML/JS boundary

When the same option is set twice, the more specific source wins. Functions can only be passed from JavaScript, which is why async fetching, render hooks and custom validators have no data-attribute equivalent.

PrioritySourceExample
1 — highestJavaScript optionszidanSelect('#x', { maxTags: 3 })
2data-zidan-options JSONdata-zidan-options='{"maxTags":3}'
3Individual data attributesdata-zidan-max-tags="4"
4 — lowestNative markup and plugin defaultsmultiple, required, maxlength

Counter detail: maxlength only supplies the default counter limit. data-zidan-counter (or the counter JS option) overrides it.

<select id="skills" class="zidan-select" multiple
        data-zidan-options='{"mode":"tags","maxTags":4,"placeholder":"Type and press Enter"}'>
  <option>Design</option>
  <option>Frontend</option>
</select>
// JS options beat the JSON attribute, which beats data-* attributes
zidanSelect('#skills', {
  mode: 'tags',
  maxTags: 2,
  validateTag: tag => tag.length <= 12 || 'Use 12 characters or fewer'
});

// Functions cannot come from HTML or JSON:
// fetchOptions, renderOption, renderSelected, renderTag, formatTag,
// validateTag, validate, validateDate, upload, callbacks, customClasses

Select

Remote paging, caching and tags

A paged fetchOptions returns { items, hasMore }. Pages are cached per query, the footer shows a load-more button, and scrolling near the bottom fetches the next page automatically. For a declarative endpoint that needs headers, extra parameters or a JSON body, see the API request lab. 

const picker = zidanSelect('#owner', {
  mode: 'combobox',
  fetchPageSize: 20,
  autoLoadMore: true,
  cacheResults: true,
  fetchOptions: async (query, signal, { page, pageSize }) => {
    const res = await fetch('/api/users?q=' + encodeURIComponent(query)
      + '&page=' + page + '&size=' + pageSize, { signal });
    const data = await res.json();
    return {
      items: data.rows.map(u => ({ value: u.id, label: u.name })),
      hasMore: data.page + 1 < data.pages
    };
  },
  onLoadMore: ({ page }) => console.log('page', page)
});

picker.loadMore();          // fetch the next page manually
picker.hasMoreResults();    // boolean
picker.getAsyncState();     // { query, page, hasMore, loading, fetchingMore, error }
picker.refreshOptions();    // clear cache and refetch page 0
picker.clearCache();        // drop cached pages only

Fields

Promise validation, phone, date and range

validate and validateDate may return a Promise. The field shows a pending state, writes the pending message into native validity so submission waits, aborts the previous request when the value changes and ignores stale resolutions. 

<!-- Static rules work as attributes -->
<input type="tel" data-zidan="input" data-zidan-label="Phone"
       data-zidan-phone="true" data-zidan-phone-country="GB"
       data-zidan-phone-format="international">

<input type="date" data-zidan="input" data-zidan-label="Delivery"
       data-zidan-disable-weekends="true"
       data-zidan-disabled-dates="2026-12-25,2027-01-01">

<input type="range" min="0" max="1000" step="50" value="400"
       data-zidan="input" data-zidan-range-variant="minimal"
       data-zidan-range-bubble="true" data-zidan-output="#adv-range-out">

<input type="range" min="0" max="30" value="5"
       data-zidan="input" data-zidan-range-bubble-mode="active"
       data-zidan-range-bubble-position="bottom" data-zidan-range-fill="false"
       data-zidan-range-min-label="Low" data-zidan-range-max-label="High"
       data-zidan-range-ticks="10,20">
const email = zidanField('#email', {
  alwaysValidate: true,
  showValidState: true,
  messages: { pending: 'Checking…', generic: 'That email is not available.' },
  validate: async (value, field, { signal }) => {
    if (!value) return true;
    const res = await fetch('/api/check-email?e=' + encodeURIComponent(value), { signal });
    const { taken } = await res.json();
    return taken ? 'This email is already registered.' : true;
  }
});

await email.validate();   // Promise<boolean>
email.isPending();        // true while the request is in flight
email.checkAsync();       // run the async rule without changing focus
email.setError('…');      // manual message
email.clearError();

// Range chrome: same options as data attributes, plus array tick values
zidanField('#adv-range-chrome', {
  rangeTicks: [10, 20],         // or '10,20' from the data attribute
  rangeBubbleMode: 'active',     // 'always' | 'active' | 'never'
  rangeBubblePosition: 'bottom', // 'top' | 'bottom'
  rangeFill: false,              // progress fill off
  rangeMinLabel: 'Low',
  rangeMaxLabel: 'High',
  rangeFormat: value => value + ' days'
});

Files

Upload pipeline: handlers, progress and retries

Native input.files stays the source of truth. Removing a file rebuilds it with DataTransfer. For real uploads use an XHR endpoint or a custom upload() handler; for demos use simulateUpload. 

zidanField('#files', {
  variant: 'dropzone',
  maxFiles: 4,
  maxSizeMB: 5,
  upload(file, context) {
    // context: { file, field, formData, signal, onProgress, complete, fail }
    fetch('/upload', {
      method: 'POST',
      body: context.formData,
      signal: context.signal
    })
      .then(res => res.ok ? res.json() : Promise.reject(res.status))
      .then(context.complete)
      .catch(context.fail);
  },
  renderFile(itemEl, { index }) {
    itemEl.dataset.index = index;
  }
});

// Or let the plugin send multipart XHR:
// zidanField('#files', { uploadUrl: '/api/files', uploadField: 'file' });

       data-zidan-max-size-mb="5"

Integration

Events and lifecycle

Events bubble from the enhanced wrapper, so one document-level listener can cover a whole form. Every listener, DOM node and in-flight request is removed by destroy().

EventDetail
zidan:change{ instance, kind, value }, { checked } or { files }
zidan:validation{ instance, state, message? }
zidan:async-validation{ valid, message } when a Promise rule settles
zidan:load-more{ page, query } for appended remote pages
zidan:step{ instance, value, direction }
zidan:phone-country / zidan:phone-format{ country } / { format }
zidan:icon-click / zidan:pickerIcon action or date/time picker request
zidan:upload-*File state, name, progress, result or message
zidan:remove / zidan:clearFile removal and field clearing
zidan:init / zidan:destroy / zidan:errorLifecycle and caught callback/upload errors
 
document.addEventListener('zidan:change', event => {
  console.log(event.detail.kind, event.detail.value || event.detail.checked);
});

document.addEventListener('zidan:upload-progress', event => {
  console.log(event.detail.name, event.detail.progress + '%');
});

const observer = ZidanForm.observe(document.body); // adds/removes nodes
// teardown a page, a route panel or one component subtree
observer.disconnect();
ZidanForm.destroy(host); // scoped: fields + selects inside host
ZidanForm.destroyAll();  // whole document

// Per instance
field.update();      // mirror native disabled/required/value
field.refresh();     // re-render without rebuilding
field.destroy();     // restore original markup + listeners

The observer reacts to nodes added or removed inside the observed root. If you remove the container itself, call field.destroy() or ZidanForm.destroyAll() first.
